import streamlit as st import streamlit.components.v1 as components import json # Configure Streamlit page settings st.set_page_config(page_title="公館晚餐吃什麼", page_icon="👾", layout="centered") # ========================================== # Inject 8-Bit Retro Pixel Art CSS (Override Streamlit defaults) # ========================================== st.markdown(""" """, unsafe_allow_html=True) # Render main application header st.title("👾 晚餐吃什麼 👾") # Initialize default food options in session state if "options" not in st.session_state: st.session_state.options = ["小高", "藏壽司", "麥當勞"] add_col, remove_col = st.columns(2) # Column for adding a new option with add_col: st.markdown("### 💾 新增選項") new_option_text = st.text_input("輸入新選項：", label_visibility="collapsed") if st.button(">> INSERT COIN / ADD <<", use_container_width=True): if new_option_text.strip() and new_option_text.strip() not in st.session_state.options: st.session_state.options.append(new_option_text.strip()) st.rerun() # Column for removing an existing option with remove_col: st.markdown("### 🗑️ 移除選項") option_to_remove = st.selectbox("選擇要移除的選項：", st.session_state.options, label_visibility="collapsed") if st.button(">> DELETE ITEM <<", use_container_width=True): if option_to_remove in st.session_state.options: st.session_state.options.remove(option_to_remove) st.rerun() # Display current inventory of food options st.markdown(f"**[ INVENTORY ({len(st.session_state.options)}) ]**： " + " // ".join(st.session_state.options)) st.markdown("---") # Render the pixel-art roulette wheel if options exist if len(st.session_state.options) > 0: options_json = json.dumps(st.session_state.options) html_content = f"""
